```csharp
using System;
using System.Security.Cryptography;
using System.Text;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Identity;
using Microsoft.EntityFrameworkCore;

public class LoginService
{
    private readonly UserManager<IdentityUser> _userManager;
    private readonly SignInManager<IdentityUser> _signInManager;

    public LoginService(UserManager<IdentityUser> userManager, SignInManager<IdentityUser> signInManager)
    {
        _userManager = userManager;
        _signInManager = signInManager;
    }

    public async Task<LoginResult> LoginAsync(string username, string password)
    {
        try
        {
            // Find user by username
            var user = await _userManager.FindByNameAsync(username);
            if (user == null)
            {
                return new LoginResult { Success = false, Message = "Invalid username or password" };
            }

            // Check if user is locked out
            if (await _userManager.IsLockedOutAsync(user))
            {
                return new LoginResult { Success = false, Message = "Account is locked due to multiple failed attempts" };
            }

            // Verify password
            var result = await _signInManager.PasswordSignInAsync(username, password, false, lockoutOnFailure: true);

            if (result.Succeeded)
            {
                return new LoginResult { Success = true, Message = "Login successful" };
            }
            else if (result.IsLockedOut)
            {
                return new LoginResult { Success = false, Message = "Account is locked due to multiple failed attempts" };
            }
            else if (result.IsNotAllowed)
            {
                return new LoginResult { Success = false, Message = "Login not allowed" };
            }
            else
            {
                return new LoginResult { Success = false, Message = "Invalid username or password" };
            }
        }
        catch (Exception ex)
        {
            return new LoginResult { Success = false, Message = "An error occurred during login" };
        }
    }
}

public class LoginResult
{
    public bool Success { get; set; }
    public string Message { get; set; }
}
```

**Alternative implementation using custom user model:**

```csharp
using System;
using System.Security.Cryptography;
using System.Text;
using System.Threading.Tasks;
using Microsoft.EntityFrameworkCore;

public class CustomLoginService
{
    private readonly AppDbContext _context;

    public CustomLoginService(AppDbContext context)
    {
        _context = context;
    }

    public async Task<LoginResult> LoginAsync(string username, string password)
    {
        try
        {
            // Find user in database
            var user = await _context.Users
                .FirstOrDefaultAsync(u => u.Username == username);

            if (user == null)
            {
                return new LoginResult { Success = false, Message = "Invalid username or password" };
            }

            // Verify password using salted hash
            if (!VerifyPassword(password, user.PasswordHash, user.Salt))
            {
                return new LoginResult { Success = false, Message = "Invalid username or password" };
            }

            // Check if account is active
            if (!user.IsActive)
            {
                return new LoginResult { Success = false, Message = "Account is deactivated" };
            }

            // Login successful
            return new LoginResult { Success = true, Message = "Login successful" };
        }
        catch (Exception ex)
        {
            return new LoginResult { Success = false, Message = "An error occurred during login" };
        }
    }

    private bool VerifyPassword(string password, string hashedPassword, string salt)
    {
        using (var sha256 = SHA256.Create())
        {
            var saltedPassword = password + salt;
            var hash = sha256.ComputeHash(Encoding.UTF8.GetBytes(saltedPassword));
            var hashString = Convert.ToBase64String(hash);

            return hashString == hashedPassword;
        }
    }
}
```

**Key features of this implementation:**

1. **Password Security**: Uses salted hashing with SHA256
2. **Account Lockout**: Prevents brute force attacks
3. **Error Handling**: Comprehensive exception handling
4. **Async Support**: Non-blocking operations
5. **Validation**: Checks for valid user and active account
6. **Return Results**: Clear success/failure indicators with messages
